// ==UserScript==
// @name         X-Precision Client V3.0 - Ultimate Edition

(function() {
    'use strict';

    // ==========================================
    // CONFIGURATION & STATE
    // ==========================================
    const Config = {
        // Combat
        autoAim: true,
        aimBotFOV: 360,
        autoShoot: true,
        autoHeal: true,
        healThreshold: 40,
        antiShame: true,
        
        // Building
        autoPlace: true,
        autoReplace: true, // Aggressively replace weak/bad structures
        replacePriority: 'spike', // Prioritize replacing with spikes
        fastBuild: true,

        // R-Target System
        rTargetEnabled: true,
        rTargetKey: 'r',
        instaKillNoHelm: true,
        instaKillOnDamage: true,

        // Smart Hats
        smartHat: true,
        empThreshold: 2, // Number of turrets to trigger EMP helm
        soldierHelmetHP: 80, // Use soldier helm if HP below this when fighting

        // Visuals
        showMenu: true,
        menuKey: 'Insert',
        drawESP: true,
        drawHealth: true
    };

    const State = {
        myPlayer: null,
        players: [],
        buildings: [],
        projectiles: [],
        targetPlayer: null, // Locked by R key
        nearbyTurrets: 0,
        nearbyEnemies: 0,
        menuOpen: false,
        lastHealTime: 0,
        lastPlaceTime: 0
    };

    // ==========================================
    // MATH & UTILS
    // ==========================================
    const MathUtils = {
        dist: (x1, y1, x2, y2) => Math.hypot(x2 - x1, y2 - y1),
        angle: (x1, y1, x2, y2) => Math.atan2(y2 - y1, x2 - x1),
        clamp: (val, min, max) => Math.min(Math.max(val, min), max),
        lerp: (start, end, amt) => (1 - amt) * start + amt * end
    };

    // ==========================================
    // GAME HOOKS & DATA PARSING
    // ==========================================
    function updateGameData(rawData) {
        // Parse raw game data to populate State.players, State.buildings, etc.
        State.nearbyTurrets = countNearbyBuildings('turret');
        State.nearbyEnemies = State.players.filter(p => p.id !== State.myPlayer?.id && !p.allied).length;
    }

    function countNearbyBuildings(type) {
        if (!State.myPlayer || !State.buildings) return 0;
        return State.buildings.filter(b => b.type === type && MathUtils.dist(State.myPlayer.x, State.myPlayer.y, b.x, b.y) < 400).length;
    }

    function getNearestEnemy() {
        if (!State.myPlayer) return null;
        let nearest = null;
        let minDst = Infinity;
        for (const p of State.players) {
            if (p.id === State.myPlayer.id || p.allied || p.dead) continue;
            const d = MathUtils.dist(State.myPlayer.x, State.myPlayer.y, p.x, p.y);
            if (d < minDst) {
                minDst = d;
                if (d < 600) nearest = p;
            }
        }
        return nearest;
    }

    // ==========================================
    // LOGIC: SMART AUTO-HAT
    // ==========================================
    function handleSmartHat() {
        if (!Config.smartHat || !State.myPlayer) return;

        const turretCount = countNearbyBuildings('turret');
        const enemy = getNearestEnemy();
        const currentHat = State.myPlayer.hat;

        // Priority 1: Turrets -> EMP Helmet
        if (turretCount >= Config.empThreshold) {
            if (currentHat !== 'emp') tryEquipItem('emp');
            return;
        }

        // Priority 2: Enemies Nearby -> Soldier Helmet (if low HP) or Barbarian/DPS
        if (enemy) {
            const distToEnemy = MathUtils.dist(State.myPlayer.x, State.myPlayer.y, enemy.x, enemy.y);
            if (distToEnemy < 300) {
                if (State.myPlayer.hp < Config.soldierHelmetHP) {
                    if (currentHat !== 'soldier') tryEquipItem('soldier');
                } else {
                    if (currentHat !== 'barbarian') tryEquipItem('barbarian');
                }
                return;
            }
        }
    }

    function tryEquipItem(itemCode) {
        // io.emit('equip', itemCode); 
    }

    // ==========================================
    // LOGIC: R-TARGET & INSTA-KILL
    // ==========================================
    function handleRTarget() {
        if (!Config.rTargetEnabled || !State.myPlayer) return;

        if (!State.targetPlayer || State.targetPlayer.dead) {
            State.targetPlayer = getNearestEnemy();
        }

        if (State.targetPlayer) {
            const dist = MathUtils.dist(State.myPlayer.x, State.myPlayer.y, State.targetPlayer.x, State.targetPlayer.y);
            
            drawTargetIndicator(State.targetPlayer);

            const shouldAttack = () => {
                if (Config.instaKillNoHelm && State.targetPlayer.hat !== 'soldier') return true;
                if (Config.instaKillOnDamage && dist < 200) return true;
                return false;
            };

            if (dist < 150 && shouldAttack()) {
                const angle = MathUtils.angle(State.myPlayer.x, State.myPlayer.y, State.targetPlayer.x, State.targetPlayer.y);
                aimAt(angle);
                shoot();
            }
        }
    }

    function drawTargetIndicator(player) {
        // Canvas drawing logic
    }

    // ==========================================
    // LOGIC: SUPERIOR AUTO-PLACE & REPLACE
    // ==========================================
    function handleAutoBuild() {
        if (!Config.autoPlace || !State.myPlayer) return;

        const pos = State.myPlayer;
        const buildPos = getOptimalBuildPosition(pos);

        if (buildPos) {
            if (Config.autoReplace) {
                const existing = getBuildingAt(buildPos.x, buildPos.y);
                if (existing) {
                    if (shouldReplace(existing)) {
                        breakBuilding(existing);
                        setTimeout(() => placeBuilding(buildPos.x, buildPos.y, Config.replacePriority), 50);
                        return;
                    }
                }
            }
            
            if (!getBuildingAt(buildPos.x, buildPos.y)) {
                placeBuilding(buildPos.x, buildPos.y, Config.replacePriority);
            }
        }
    }

    function getOptimalBuildPosition(player) {
        const enemy = getNearestEnemy();
        if (!enemy) return null;

        const angleToEnemy = MathUtils.angle(player.x, player.y, enemy.x, enemy.y);
        const dist = 40;
        
        const targetX = player.x + Math.cos(angleToEnemy) * dist;
        const targetY = player.y + Math.sin(angleToEnemy) * dist;

        return { x: targetX, y: targetY };
    }

    function shouldReplace(building) {
        if (Config.replacePriority === 'spike' && building.type !== 'spike') return true;
        if (building.hp < 20) return true;
        return false;
    }

    function breakBuilding(building) {
        // io.emit('hit', building.id);
    }

    function placeBuilding(x, y, type) {
        // io.emit('place', { x, y, type });
    }

    function getBuildingAt(x, y) {
        return State.buildings.find(b => MathUtils.dist(b.x, b.y, x, y) < 10);
    }

    // ==========================================
    // LOGIC: AUTO-HEAL & ANTI-SHAME
    // ==========================================
    function handleSurvival() {
        if (!State.myPlayer) return;

        if (Config.autoHeal && State.myPlayer.hp < Config.healThreshold) {
            const now = Date.now();
            if (now - State.lastHealTime > 1000) {
                useFood();
                State.lastHealTime = now;
            }
        }

        if (Config.antiShame) {
            detectAndBreakShame();
        }
    }

    function detectAndBreakShame() {
        const surroundingWalls = State.buildings.filter(b => {
            const d = MathUtils.dist(State.myPlayer.x, State.myPlayer.y, b.x, b.y);
            return d < 30 && d > 10 && (b.type === 'wall' || b.type === 'spike');
        });

        if (surroundingWalls.length >= 4) {
            const nearestWall = surroundingWalls.sort((a, b) => 
                MathUtils.dist(State.myPlayer.x, State.myPlayer.y, a.x, a.y) - 
                MathUtils.dist(State.myPlayer.x, State.myPlayer.y, b.x, b.y)
            )[0];
            
            if (nearestWall) {
                breakBuilding(nearestWall);
                const angle = MathUtils.angle(nearestWall.x, nearestWall.y, State.myPlayer.x, State.myPlayer.y);
                moveInDirection(angle);
            }
        }
    }

    function useFood() { /* io.emit('food'); */ }
    function aimAt(angle) { /* mouse.angle = angle; */ }
    function shoot() { /* io.emit('shoot'); */ }
    function moveInDirection(angle) { /* Set movement keys */ }

    // ==========================================
    // INPUT HANDLING
    // ==========================================
    window.addEventListener('keydown', (e) => {
        if (e.key.toLowerCase() === Config.rTargetKey) {
            State.targetPlayer = getNearestEnemy();
            if (State.targetPlayer) console.log("Target Locked:", State.targetPlayer.name);
        }
        if (e.key === Config.menuKey) toggleMenu();
    });

    // ==========================================
    // MENU SYSTEM
    // ==========================================
    function createMenu() {
        const menu = document.createElement('div');
        menu.id = 'xPrecisionMenu';
        menu.style.cssText = `
            position: fixed; top: 10px; left: 10px; width: 280px; 
            background: rgba(0, 0, 0, 0.85); color: #0f0; font-family: monospace; 
            padding: 15px; border-radius: 8px; border: 1px solid #0f0; 
            z-index: 9999; display: none; user-select: none;
            box-shadow: 0 0 15px rgba(0, 255, 0, 0.3);
        `;
        
        const title = document.createElement('h3');
        title.innerText = 'X-PRECISION V3.0';
        title.style.textAlign = 'center';
        title.style.marginTop = '0';
        title.style.color = '#fff';
        title.style.textShadow = '0 0 5px #0f0';
        menu.appendChild(title);

        function addToggle(label, configKey) {
            const row = document.createElement('div');
            row.style.marginBottom = '8px';
            row.style.display = 'flex';
            row.style.justifyContent = 'space-between';
            row.style.alignItems = 'center';
            
            const lbl = document.createElement('span');
            lbl.innerText = label;
            
            const chk = document.createElement('input');
            chk.type = 'checkbox';
            chk.checked = Config[configKey];
            chk.style.cursor = 'pointer';
            chk.onchange = (e) => Config[configKey] = e.target.checked;

            row.appendChild(lbl);
            row.appendChild(chk);
            menu.appendChild(row);
        }

        function addSlider(label, configKey, min, max, step) {
            const row = document.createElement('div');
            row.style.marginBottom = '8px';
            
            const lbl = document.createElement('div');
            lbl.innerText = `${label}: ${Config[configKey]}`;
            lbl.style.fontSize = '12px';
            
            const rng = document.createElement('input');
            rng.type = 'range';
            rng.min = min; rng.max = max; rng.step = step;
            rng.value = Config[configKey];
            rng.style.width = '100%';
            rng.oninput = (e) => {
                Config[configKey] = parseFloat(e.target.value);
                lbl.innerText = `${label}: ${Config[configKey]}`;
            };

            row.appendChild(lbl);
            row.appendChild(rng);
            menu.appendChild(row);
        }

        // Combat Section
        const sec1 = document.createElement('div');
        sec1.style.borderBottom = '1px solid #333';
        sec1.style.marginBottom = '10px';
        sec1.style.paddingBottom = '10px';
        sec1.innerHTML = '<strong style="color:#fff">COMBAT</strong>';
        menu.appendChild(sec1);
        
        addToggle('Auto Aim', 'autoAim');
        addToggle('Auto Shoot', 'autoShoot');
        addToggle('Auto Heal', 'autoHeal');
        addSlider('Heal Threshold (%)', 'healThreshold', 0, 100, 5);
        addToggle('Anti-Shame', 'antiShame');

        // Building Section
        const sec2 = document.createElement('div');
        sec2.style.borderBottom = '1px solid #333';
        sec2.style.marginBottom = '10px';
        sec2.style.paddingBottom = '10px';
        sec2.innerHTML = '<strong style="color:#fff">BUILDING</strong>';
        menu.appendChild(sec2);

        addToggle('Auto Place', 'autoPlace');
        addToggle('Aggressive Replace', 'autoReplace');
        addToggle('Fast Build', 'fastBuild');

        // R-Target Section
        const sec3 = document.createElement('div');
        sec3.style.borderBottom = '1px solid #333';
        sec3.style.marginBottom = '10px';
        sec3.style.paddingBottom = '10px';
        sec3.innerHTML = '<strong style="color:#f55">R-TARGET SYSTEM</strong>';
        menu.appendChild(sec3);
        
        const rDesc = document.createElement('div');
        rDesc.innerText = 'Press "R" to lock nearest enemy.';
        rDesc.style.fontSize = '10px';
        rDesc.style.color = '#aaa';
        rDesc.style.marginBottom = '5px';
        menu.appendChild(rDesc);

        addToggle('Enable R-Target', 'rTargetEnabled');
        addToggle('Insta-Kill (No Helm)', 'instaKillNoHelm');
        addToggle('Insta-Kill (On Dmg)', 'instaKillOnDamage');

        // Smart Hats Section
        const sec4 = document.createElement('div');
        sec4.style.borderBottom = '1px solid #333';
        sec4.style.marginBottom = '10px';
        sec4.style.paddingBottom = '10px';
        sec4.innerHTML = '<strong style="color:#fff">SMART HATS</strong>';
        menu.appendChild(sec4);

        addToggle('Smart Hat Switching', 'smartHat');
        addSlider('EMP Trigger (Turrets)', 'empThreshold', 1, 5, 1);
        addSlider('Soldier Helm HP%', 'soldierHelmetHP', 0, 100, 5);

        document.body.appendChild(menu);
    }

    function toggleMenu() {
        const menu = document.getElementById('xPrecisionMenu');
        if (!menu) {
            createMenu();
            toggleMenu();
            return;
        }
        Config.menuOpen = !Config.menuOpen;
        menu.style.display = Config.menuOpen ? 'block' : 'none';
    }

    // ==========================================
    // MAIN LOOP
    // ==========================================
    function gameLoop() {
        handleSmartHat();
        handleRTarget();
        handleAutoBuild();
        handleSurvival();
        requestAnimationFrame(gameLoop);
    }

    console.log('X-Precision Client V3.0 Loaded');
    createMenu();
    gameLoop();

})();
